cmake_minimum_required(VERSION 3.20)
project(go2_stop_envelope VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(stop_model INTERFACE)
target_include_directories(stop_model INTERFACE cpp/include)
add_executable(stop-envelope cpp/src/main.cpp)
target_link_libraries(stop-envelope PRIVATE stop_model)
add_executable(stop-model-tests cpp/tests/test_model.cpp)
target_link_libraries(stop-model-tests PRIVATE stop_model)
foreach(target stop-envelope stop-model-tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME stop-model COMMAND stop-model-tests)
